## Auth 설정(src/lib/auth.ts)

```ts
import NextAuth, { type NextAuthOptions } from 'next-auth'
import Credentials from 'next-auth/providers/credentials'
import { prisma } from './prisma'
import bcrypt from 'bcryptjs'

export const authOptions: NextAuthOptions = {
  session: { strategy: 'jwt' },
  pages: { signIn: '/ko/auth/signin' },
  providers: [
    Credentials({
      name: 'Credentials',
      credentials: { email: {}, password: {} },
      async authorize(creds) {
        if (!creds?.email || !creds?.password) return null
        const user = await prisma.user.findUnique({ where: { email: creds.email } })
        if (!user) return null
        const ok = await bcrypt.compare(creds.password, user.password)
        return ok ? { id: user.id, email: user.email, name: user.name, role: user.role } as any : null
      }
    })
  ],
  callbacks: {
    async jwt({ token, user }) {
      if (user) token.role = (user as any).role
      return token
    },
    async session({ session, token }) {
      (session as any).role = token.role
      return session
    }
  }
}
```

```ts
  const url = process.env.SLACK_WEBHOOK_URL
  if (!url) throw new Error('SLACK_WEBHOOK_URL not set')
```

## 문의 API(src/app/api/contact/route.ts)

```ts
import { NextResponse } from 'next/server'
import { z } from 'zod'
import { sendToSlack } from '@/lib/slack'

const Schema = z.object({
  name: z.string().min(1),
  phone: z.string().min(7),
  email: z.string().email(),
  budget: z.string().optional(),
  message: z.string().min(1)
})

export async function POST(req: Request) {
  const json = await req.json()
  const data = Schema.parse(json)
  await sendToSlack({
    text: `🆕 프로젝트 문의\n• 이름: ${data.name}\n• 연락처: ${data.phone}\n• 이메일: ${data.email}\n• 예산: ${data.budget ?? '-'}\n• 내용: ${data.message}`
  })
  return NextResponse.json({ ok: true })
}
```

## 프로젝트 문의 폼(src/app/(public)/\[locale]/contact/page.tsx)

```tsx
'use client'
import { useState } from 'react'

export default function Contact() {
  const [state, setState] = useState<'idle'|'sending'|'sent'|'error'>('idle')
  async function onSubmit(e: any) {
    e.preventDefault(); setState('sending')
    const body = Object.fromEntries(new FormData(e.currentTarget).entries())
    try {
      const res = await fetch('/api/contact', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) })
      if (!res.ok) throw new Error('fail')
      setState('sent'); e.currentTarget.reset()
    } catch { setState('error') }
  }
  return (
    <div>
      <h1 className="text-3xl font-semibold mb-6">프로젝트 문의</h1>
      <form onSubmit={onSubmit} className="grid gap-3 max-w-xl">
        <input name="name" placeholder="이름" className="border p-3 rounded-xl" required />
        <input name="phone" placeholder="연락처" className="border p-3 rounded-xl" required />
        <input name="email" type="email" placeholder="이메일" className="border p-3 rounded-xl" required />
        <input name="budget" placeholder="예산 범위(선택)" className="border p-3 rounded-xl" />
        <textarea name="message" placeholder="요청사항" className="border p-3 rounded-xl" required />
        <button className="bg-black text-white px-5 py-3 rounded-2xl" disabled={state==='sending'}>
          {state==='sending' ? '전송 중…' : '보내기'}
        </button>
        {state==='sent' && <p className="text-green-600">전송되었습니다. 곧 연락드릴게요.</p>}
        {state==='error' && <p className="text-red-600">전송 오류가 발생했습니다.</p>}
      </form>
    </div>
  )
}
```

## Admin 보호 레이아웃(src/app/(protected)/layout.tsx)

```tsx
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import Link from 'next/link'

export default async function ProtectedLayout({ children }: any) {
  const session = await getServerSession(authOptions)
  if (!session || (session as any).role !== 'ADMIN') {
```

```ts
import { prisma } from '@/lib/prisma'
import { NextResponse } from 'next/server'

export async function GET() {
  const list = await prisma.blog.findMany({ orderBy: { createdAt: 'desc' } })
  return NextResponse.json(list)
}

export async function POST(req: Request) {
  const data = await req.json()
```

## 배포/운영 가이드 (Replit)

1. **Replit → Create Repl → Node.js** 대신 **Next.js** 템플릿 선택
2. 위 파일 구조대로 생성 후 패키지 설치: `npm i`
3. `npx prisma generate` → `npx prisma migrate dev --name init`
4. `.env` 작성 (Webhook/도메인 등)
5. "Run" 시 포트 3000 노출, Replit 제공 URL 확인
6. 커스텀 도메인 연결: `www.xn--oi2ba04bai4alco.co.kr`(국문 도메인 punycode) CNAME → Replit 설정

**시드 스니펫(관리자 생성)**

```ts
// scripts/seed-admin.ts (임시 실행: ts-node 또는 node --loader ts-node/esm)
import { prisma } from '@/lib/prisma'
import bcrypt from 'bcryptjs'

async function main(){
  const email = 'admin@beaulead.io'
  const password = await bcrypt.hash('ChangeMe123!', 10)
  await prisma.user.upsert({
    where: { email },
    update: { role: 'ADMIN' },
    create: { email, password, role: 'ADMIN', name: 'Admin' }
  })
  console.log('Admin ready')
}
main()
```
