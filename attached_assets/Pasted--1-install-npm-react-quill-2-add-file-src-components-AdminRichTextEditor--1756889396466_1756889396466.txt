# 1) 패키지 설치
install npm: react-quill

# 2) 에디터 컴포넌트 생성
add file: src/components/AdminRichTextEditor.jsx
with content:
"import React from 'react';
import ReactQuill from \"react-quill\";
import \"react-quill/dist/quill.snow.css\";
import \"../styles/editor.css\";

export default function AdminRichTextEditor({ value, onChange, placeholder }) {
  const modules = {
    toolbar: [
      [{ header: [1, 2, 3, false] }],
      [{ font: [] }, { size: [] }],
      ['bold','italic','underline','strike'],
      [{ color: [] }, { background: [] }],
      [{ list: 'ordered' }, { list: 'bullet' }, { align: [] }],
      ['link','image','blockquote','code-block','clean'],
    ],
  };
  const formats = [
    'header','font','size','bold','italic','underline','strike',
    'color','background','list','bullet','align','link','image','blockquote','code-block'
  ];

  return (
    <div className=\"admin-editor-wrapper not-prose\">
      <ReactQuill
        theme=\"snow\"
        value={value}
        onChange={onChange}
        modules={modules}
        formats={formats}
        placeholder={placeholder}
      />
    </div>
  );
}
"
