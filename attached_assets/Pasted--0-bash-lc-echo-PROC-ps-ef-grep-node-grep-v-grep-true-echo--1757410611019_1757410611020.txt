# 3) 흔한 원인 즉시 조치
#  - dev 프리뷰에서 뜨는 디버그 alert 억제
#  - service worker가 캐시를 고정해두었을 경우 비활성화
apply_patch << 'PATCH'
*** Begin Patch
*** Update File: client/src/main.tsx
@@
 import React from 'react'
 import ReactDOM from 'react-dom/client'
 import App from './App.tsx'
 import './index.css'
+// --- DEV 프리뷰 UX 안정화: 디버그 alert 억제 ---
+if (import.meta.env.DEV) {
+  const orig = window.alert?.bind(window) as any;
+  window.alert = (msg?: any) => console.info('[dev alert suppressed]', msg);
+}
*** End Patch
PATCH

# (서비스워커가 있다면 등록 해제하도록 가드 추가)
apply_patch << 'PATCH'
*** Begin Patch
*** Update File: client/src/App.tsx
@@
 import React from 'react'
+// 서비스워커 제거(존재 시) - dev 프리뷰 캐시 고착 방지
+if ('serviceWorker' in navigator) {
+  navigator.serviceWorker.getRegistrations?.().then(regs => regs.forEach(r => r.unregister()));
+}
*** End Patch
PATCH

# 4) 클린 빌드 → 재시작 (빌드ID 스탬프 포함)
bash -lc '
rm -rf dist client/dist || true
BUILD_ID=$(date +%Y%m%d-%H%M%S)-$(git rev-parse --short HEAD 2>/dev/null || echo dev)
mkdir -p dist/public client/dist
echo "$BUILD_ID" | tee dist/public/build-id.txt client/dist/build-id.txt >/dev/null
npm ci --silent || npm i --silent
npm run build --silent || true
pkill -f "node" || true
npm start --silent & sleep 3
echo; echo "=== /api/version ==="; curl -sS http://localhost:$PORT/api/version || true
'
