# 2) 서버 캐치올 재확인: SPA에서 /inquiry도 index.html 반환 (이미 설정되어도 안전)
#  - /api, /uploads 제외 모든 경로는 index.html을 주도록 되어 있어야 합니다.
apply_patch << 'PATCH'
*** Begin Patch
*** Update File: server/index.ts
@@
     // 업로드/API가 아닌 모든 경로는 SPA index.html 반환
     app.get("*", (req: Request, res: Response, next: NextFunction) => {
       if (req.path.startsWith("/api") || req.path.startsWith("/uploads")) return next();
       if (fs.existsSync(CLIENT_INDEX_HTML)) {
         // HTML은 항상 최신으로
         res.setHeader("Cache-Control", "no-store");
         res.setHeader("X-Build-Id", readBuildId());
         return res.sendFile(CLIENT_INDEX_HTML);
       }
       return next();
     });
*** End Patch
PATCH
