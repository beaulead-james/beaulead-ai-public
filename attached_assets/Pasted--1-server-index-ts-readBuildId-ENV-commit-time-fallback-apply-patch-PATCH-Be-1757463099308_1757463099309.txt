# 4) 클린 빌드 & 재기동 (빌드 아이디 파일이 있으면 사용, 없으면 fallback이 동작)
bash -lc '
rm -rf dist client/dist || true
BUILD_ID=$(date +%Y%m%d-%H%M%S)-$(git rev-parse --short HEAD 2>/dev/null || echo dev)
mkdir -p dist/public client/dist
echo "$BUILD_ID" | tee dist/public/build-id.txt client/dist/build-id.txt >/dev/null
npm ci --silent || npm i --silent
npm run build --silent
pkill -f node || true
npm start --silent & sleep 3
'
