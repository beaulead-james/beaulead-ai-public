작업명: 회사소개 > 대표인사말 페이지 생성 (KO/EN, SEO 포함)

목표:
- 경로: /about/ceo (KO), /en/about/ceo (EN)
- 벤치마킹 톤: 히어로(타이틀+서브카피) → CEO 소개(사진/이름/직함/서명) → 메시지(문단) → 핵심가치(3~4개 카드) → CTA(프로젝트문의/포트폴리오)
- SEO 메타 및 구조화 데이터(Person) 추가
- 내비게이션에 "대표인사말" 링크 추가

5) 페이지 컴포넌트 생성: client/src/pages/about/CeoGreeting.jsx
import React from "react";
import { Helmet } from "react-helmet-async";
import { useTranslation } from "react-i18next";
import "../../styles/ceo.css";

export default function CeoGreeting({ locale="ko" }) {
  const { t, i18n } = useTranslation();
  if (i18n.language !== locale) i18n.changeLanguage(locale);

  const jsonLd = {
    "@context": "https://schema.org",
    "@type": "Person",
    "name": t("ceo.name"),
    "jobTitle": t("ceo.role"),
    "worksFor": { "@type": "Organization", "name": "뷰리드AI" }
  };

  return (
    <div className="container">
      <Helmet>
        <title>{t("ceo.title")} | 뷰리드AI</title>
        <meta name="description" content={t("ceo.heroSub")} />
        <script type="application/ld+json">{JSON.stringify(jsonLd)}</script>
      </Helmet>

      <section className="ceo-hero">
        <h1 className="ceo-title">{t("ceo.heroTitle")}</h1>
        <p className="ceo-sub">{t("ceo.heroSub")}</p>
        <div className="cta">
          <a className="btn primary" href="/contact">{t("ceo.ctaConsult")}</a>
          <a className="btn" href="/portfolio">{t("ceo.ctaPortfolio")}</a>
        </div>
      </section>

      <section className="section ceo-card">
        <img className="ceo-photo" src="/assets/ceo.jpg" alt="CEO" />
        <div>
          <div style={{fontWeight:700}}>{t("ceo.name")}</div>
          <div style={{opacity:.8,marginTop:4}}>{t("ceo.role")}</div>
        </div>
      </section>

      <section className="section">
        <p>{t("ceo.messageP1")}</p>
        <p style={{marginTop:12}}>{t("ceo.messageP2")}</p>
      </section>

      <section className="section">
        <h3 style={{fontWeight:700, marginBottom:8}}>{t("ceo.valuesTitle")}</h3>
        <div className="values">
          {t("ceo.values", { returnObjects: true }).map((v, i) => (
            <div key={i} className="value">
              <div style={{fontWeight:700}}>{v.title}</div>
              <div style={{opacity:.85, marginTop:6}}>{v.desc}</div>
            </div>
          ))}
        </div>
      </section>
    </div>
  );
}
