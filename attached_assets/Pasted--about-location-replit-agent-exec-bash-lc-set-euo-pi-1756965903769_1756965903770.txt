replit agent exec -- bash -lc '
set -euo pipefail

# 0) 위치 페이지 파일 경로 지정(없으면 생성)
PAGE_DIR="src/pages/about"
PAGE_FILE="$PAGE_DIR/Location.tsx"
mkdir -p "$PAGE_DIR"

echo "== 1) Location 페이지 컴포넌트 생성/갱신 =="
cat > "$PAGE_FILE" <<'"EOF"'
import React from "react";

/**
 * ✅ 위치 데이터 스키마 표준화 어댑터
 * - 백엔드/기존 JSON이 제각각이어도 이 함수가 단일 스키마로 변환
 * - 누락 필드는 안전한 기본값으로 대체해 렌더 오류(배열/키 불일치) 방지
 */
type RawBranch = any;
type Branch = {
  id: string;
  name: string;
  address: string;
  phone?: string;
  hours?: string;
  note?: string;
  lat?: number;
  lng?: number;
  image?: string;
};

function toBranch(raw: RawBranch, idx: number): Branch {
  const coerceNum = (v: any) => {
    const n = typeof v === "string" ? parseFloat(v) : v;
    return Number.isFinite(n) ? n : undefined;
  };
  return {
    id: String(raw?.id ?? raw?.uid ?? raw?.slug ?? `branch-${idx}`),
    name: String(raw?.name ?? raw?.title ?? raw?.branchName ?? "지점"),
    address: String(raw?.address ?? raw?.addr ?? raw?.location ?? "주소 정보 준비중"),
    phone: raw?.phone ?? raw?.tel ?? raw?.contact ?? "",
    hours: raw?.hours ?? raw?.time ?? raw?.businessHours ?? "",
    note: raw?.note ?? raw?.desc ?? raw?.description ?? "",
    lat: coerceNum(raw?.lat ?? raw?.latitude),
    lng: coerceNum(raw?.lng ?? raw?.longitude),
    image: raw?.image ?? raw?.thumbnail ?? raw?.img ?? "",
  };
}

/**
 * ✅ 실제 데이터 소스:
 * - 우선 window.__BEAULEAD_BRANCHES 또는 임포트된 상수 사용
 * - 아무 것도 없으면 샘플 1개로 폴백 (UI 깨짐 방지)
 */
declare global {
  interface Window { __BEAULEAD_BRANCHES?: RawBranch[] }
}
const fromWindow: RawBranch[] | undefined = typeof window !== "undefined" ? window.__BEAULEAD_BRANCHES : undefined;

// 필요 시 외부에서 import 해올 수 있음 (예: src/data/branches.ts)
// import { BRANCHES as Imported } from "../../data/branches";

const RAW: RawBranch[] = (fromWindow && Array.isArray(fromWindow) && fromWindow.length > 0)
  ? fromWindow
  // : (Imported ?? [])  // ← 실제 데이터 파일 있으면 주석 해제
  : [
      {
        id: "hq",
        name: "뷰리드AI 강남 본사",
        address: "서울특별시 강남구 테헤란로 000",
        phone: "02-000-0000",
        hours: "평일 10:00–19:00",
        note: "방문 전 예약 부탁드립니다.",
        lat: 37.4979,
        lng: 127.0276,
        image: "",
      },
    ];

const BRANCHES: Branch[] = RAW.map(toBranch);

const SectionTitle: React.FC<{title: string; desc?: string}> = ({title, desc}) => (
  <div className="mb-6">
    <h2 className="text-2xl md:text-3xl font-semibold tracking-tight">{title}</h2>
    {desc && <p className="text-sm md:text-base text-gray-500 mt-1">{desc}</p>}
  </div>
);

const InfoRow: React.FC<{label: string; value?: string}> = ({label, value}) => {
  if (!value) return null;
  return (
    <div className="flex items-start gap-2 text-sm md:text-base">
      <div className="min-w-16 text-gray-500">{label}</div>
      <div className="font-medium">{value}</div>
    </div>
  );
};

const BranchCard: React.FC<{b: Branch; onFocus: () => void; active?: boolean}> = ({b, onFocus, active}) => {
  const mapQuery = encodeURIComponent(b.address || b.name);
  return (
    <button
      onClick={onFocus}
      className={[
        "text-left w-full rounded-2xl border bg-white hover:shadow-md transition-shadow",
        active ? "border-gray-900 shadow-md" : "border-gray-200"
      ].join(" ")}
    >
      {/* 이미지 */}
      <div className="aspect-[16/9] w-full overflow-hidden rounded-t-2xl bg-gray-50">
        {b.image
          ? <img src={b.image} alt={b.name} className="h-full w-full object-cover" loading="lazy" />
          : <div className="h-full w-full flex items-center justify-center text-gray-400">No Image</div>
        }
      </div>

      {/* 텍스트 */}
      <div className="p-4 md:p-5">
        <div className="flex items-center justify-between gap-2">
          <h3 className="text-lg md:text-xl font-semibold">{b.name}</h3>
          <span className="text-xs md:text-sm px-2.5 py-1 rounded-full bg-gray-100 text-gray-600">
            위치 안내
          </span>
        </div>
        <div className="mt-3 space-y-2">
          <InfoRow label="주소" value={b.address} />
          <InfoRow label="전화" value={b.phone} />
          <InfoRow label="영업" value={b.hours} />
          {b.note && <p className="text-sm text-gray-500">{b.note}</p>}
        </div>

        {/* 작은 지도 링크 */}
        <div className="mt-4">
          <a
            href={`https://www.google.com/maps/search/?api=1&query=${mapQuery}`}
            target="_blank"
            rel="noreferrer"
            className="inline-flex items-center gap-2 text-sm underline underline-offset-4 hover:no-underline"
          >
            Google 지도로 보기 →
          </a>
        </div>
      </div>
    </button>
  );
};

const MapPanel: React.FC<{b?: Branch}> = ({b}) => {
  const center = b?.lat && b?.lng
    ? `${b.lat},${b.lng}`
    : encodeURIComponent(b?.address || "서울 강남역");
  const q = b?.name ? encodeURIComponent(`${b.name} ${b.address}`) : encodeURIComponent(b?.address || "강남역");
  const src = `https://www.google.com/maps/embed/v1/search?key=AIzaSyD-FAKE-KEY&center=${center}&q=${q}&zoom=15`;
  // ↑ 실제 구글맵 Embed API 키가 없다면 아래 fallback iframe 사용 (키 불필요)
  const fallback = `https://maps.google.com/maps?q=${q}&t=&z=15&ie=UTF8&iwloc=&output=embed`;
  return (
    <div className="rounded-2xl overflow-hidden border border-gray-200 bg-white">
      <div className="h-[320px] md:h-[480px]">
        <iframe
          title="map"
          className="w-full h-full"
          loading="lazy"
          referrerPolicy="no-referrer-when-downgrade"
          src={fallback}
        />
      </div>
      <div className="p-4 md:p-5 border-t border-gray-100">
        <SectionTitle title="방문 안내" desc="찾아오시는 길 및 주차/엘리베이터 위치를 확인하세요." />
        {b ? (
          <div className="space-y-2 text-sm md:text-base">
            <InfoRow label="지점" value={b.name} />
            <InfoRow label="주소" value={b.address} />
            <InfoRow label="전화" value={b.phone} />
            {b.note && <p className="text-gray-500">{b.note}</p>}
          </div>
        ) : (
          <p className="text-gray-500">좌측 목록에서 지점을 선택하세요.</p>
        )}
      </div>
    </div>
  );
};

const LocationPage: React.FC = () => {
  const [activeIdx, setActiveIdx] = React.useState(0);
  const active = BRANCHES[activeIdx];

  return (
    <main className="min-h-screen bg-gray-50">
      {/* Hero */}
      <section className="px-4 md:px-8 lg:px-10 pt-10 pb-6 max-w-7xl mx-auto">
        <h1 className="text-3xl md:text-4xl font-bold tracking-tight">오시는 길</h1>
        <p className="text-gray-500 mt-2">지점 정보를 한눈에 보고, 바로 길찾기를 시작하세요.</p>
      </section>

      {/* Content: 좌측 카드 그리드 + 우측 맵 패널 */}
      <section className="px-4 md:px-8 lg:px-10 pb-14 max-w-7xl mx-auto">
        <div className="grid grid-cols-1 lg:grid-cols-12 gap-6 lg:gap-8">
          {/* 목록 */}
          <div className="lg:col-span-7">
            <SectionTitle title="지점 목록" desc="카드를 클릭하면 오른쪽 지도가 해당 지점으로 이동합니다." />
            <div className="grid grid-cols-1 sm:grid-cols-2 gap-5">
              {BRANCHES.map((b, i) => (
                <BranchCard key={b.id} b={b} active={i===activeIdx} onFocus={() => setActiveIdx(i)} />
              ))}
            </div>
          </div>

          {/* 맵/상세 (데스크톱에서 sticky) */}
          <div className="lg:col-span-5">
            <div className="lg:sticky lg:top-6">
              <MapPanel b={active} />
            </div>
          </div>
        </div>
      </section>
    </main>
  );
};

export default LocationPage;
"EOF"

echo "== 2) 라우터에 /about/location 경로 연결 =="
# 라우터 파일 탐색
ROUTER_FILE="$(rg -l \"BrowserRouter|createBrowserRouter|Routes|<Router|react-router-dom|addRoute\" src | head -n1 || true)"
if [ -z "$ROUTER_FILE" ]; then
  # App.tsx에 직접 Routes가 있을 수 있음
  ROUTER_FILE="$(rg -l \"<Routes|<Route\" src/App.tsx src/main.tsx 2>/dev/null | head -n1 || true)"
fi

if [ -n "$ROUTER_FILE" ]; then
  echo " - router file: $ROUTER_FILE"
  # import 추가(중복 방지)
  if ! rg -n "Location.tsx|from \\"./pages/about/Location\\"" "$ROUTER_FILE" >/dev/null 2>&1; then
    perl -0777 -i -pe "s@(\n\s*import [^\n]+;\s*)@\1import LocationPage from \\"./pages/about/Location\\";\n@" "$ROUTER_FILE" || true
  fi
  # Route 주입 (이미 있으면 스킵)
  if rg -n "/about/location" "$ROUTER_FILE" >/dev/null 2>&1; then
    echo " - /about/location 라우트가 이미 존재합니다."
  else
    # 가장 마지막 Route 앞 또는 </Routes> 직전에 삽입
    perl -0777 -i -pe "s@(\</Routes\>)@  <Route path=\\"/about/location\\" element={<LocationPage />} />\n\1@" "$ROUTER_FILE" || true
    perl -0777 -i -pe "s@(\]\))@  , { path: \\"/about/location\\", element: <LocationPage /> }\n\1@" "$ROUTER_FILE" || true
  fi
else
  echo " - 경고: 라우터 파일을 찾지 못했습니다. App.tsx에 수동으로 LocationPage를 연결해주세요."
fi

npm run build || echo "(경고) build 스크립트 없음 — 스킵"
