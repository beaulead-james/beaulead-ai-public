2) 서버: 인증 미들웨어와 상태 확인 라우트 추가
- server/middlewares/auth.ts (신규) 생성

import { Request, Response, NextFunction } from "express";

export function requireAuth(req: Request, res: Response, next: NextFunction) {
  // @ts-ignore
  if (!req.session || !req.session.userId) {
    return res.status(401).json({ message: "Unauthorized" });
  }
  next();
}

export function requireRole(...roles: string[]) {
  return (req: Request, res: Response, next: NextFunction) => {
    // @ts-ignore
    const role = req.session?.role;
    if (!role) return res.status(401).json({ message: "Unauthorized" });
    if (!roles.includes(role) && role !== "ADMIN") {
      return res.status(403).json({ message: "Content management access required" });
    }
    next();
  };
}

- server/routes/auth.ts에 상태 조회(me)와 기존 로그인/로그아웃 노출 확인

import { Router } from "express";
const router = Router();

router.get("/me", (req, res) => {
  // @ts-ignore
  const sess = req.session || {};
  // userId/role 이 없으면 비로그인
  if (!sess.userId) return res.status(200).json({ authenticated: false });
  return res.status(200).json({
    authenticated: true,
    userId: sess.userId,
    role: sess.role
  });
});

// (이미 구현되어 있겠지만) 로그인 시:

  // @ts-ignore

  res.clearCookie("auth", { httpOnly: true, sameSite: "lax", secure: true });
  res.status(204).end();
});

export default router;

export async function createBlog(payload: any) {
  const res = await fetch("/api/blogs", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(payload),
    credentials: "include"    // ★ 쿠키 포함 필수
  });
  if (!res.ok) {
    const txt = await res.text();
    throw new Error(res.status + ": " + txt);
  }
  return res.json();
}
