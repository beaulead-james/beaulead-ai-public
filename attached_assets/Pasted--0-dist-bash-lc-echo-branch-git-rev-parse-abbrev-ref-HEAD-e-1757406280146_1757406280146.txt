# 2) 클린 빌드 (산출물 싹 갈아끼우기)
bash -lc '
rm -rf dist client/dist || true
BUILD_ID=$(date +%Y%m%d-%H%M%S)-$(git rev-parse --short HEAD)
echo "$BUILD_ID" | tee dist/public/build-id.txt client/dist/build-id.txt >/dev/null 2>&1 || true
npm ci --silent || npm i --silent
npm run build --silent
'
