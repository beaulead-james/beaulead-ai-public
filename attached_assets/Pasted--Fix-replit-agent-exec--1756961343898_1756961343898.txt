# ⛳ 복붙실행: 운영 로그인 실패 원클릭 Fix (쿠키/프록시/시크릿/권한/재배포/헬스체크)
# - 운영 도메인/관리자 계정은 필요 시 변경
replit agent exec -- bash -lc '
set -euo pipefail

PROD_DOMAIN="https://byurideu-ai-james1668.replit.app"
ADMIN_EMAIL="admin@beaulead.co.kr"
PROD_DOMAIN="https://byurideu-ai-james1668.replit.app"
ADMIN_EMAIL="admin@beaulead.co.kr"
ADMIN_PASSWORD="admin123"   # ✅ 운영 관리자 비밀번호

echo "== 0) 환경 점검 ================================"
printenv | egrep -i "NODE_ENV|DATABASE_URL|SESSION_SECRET|JWT_SECRET|CORS_ORIGIN|AUTH_ORIGIN" || true
command -v rg >/dev/null 2>&1 || { echo "ripgrep(rg)가 필요합니다. 설치 후 재실행하세요."; exit 1; }

echo "== 1) trust proxy 설정 추가 ===================="
# express 앱 생성 파일 자동 탐색 (가장 가능성 높은 파일)
SERVER_FILE="$(rg -l \"express\\(\\)\" src | head -n1 || true)"
if [ -z \"$SERVER_FILE\" ]; then
  # 넓게 탐색
  SERVER_FILE="$(rg -l \"express\\(\\)\" | head -n1 || true)"
fi

if [ -n \"$SERVER_FILE\" ]; then
  if rg -n \"app.set\\(\\x27trust proxy\\x27\" \"$SERVER_FILE\" >/dev/null 2>&1; then
    echo \" - 이미 trust proxy 설정 있음: $SERVER_FILE\"
  else
    echo \" - trust proxy 주입: $SERVER_FILE\"
    perl -0777 -i -pe \"s/const\\s+app\\s*=\\s*express\\(\\);\\n/const app = express();\\napp.set(\\x27trust proxy\\x27, 1);\\n/s\" \"$SERVER_FILE\" || true
  fi
else
  echo \" - 경고: express 앱 파일을 찾지 못했습니다. (수동 확인 필요)\"
fi

echo \"== 2) 세션/쿠키 옵션 보정 (secure, sameSite) ======\"
# 세션/쿠키 설정 파일 후보 자동 탐색
SESSION_FILE=\"\"
for f in \
  src/middleware/session.ts \
  src/middleware/session.js \
  src/server/session.ts \
  src/server/session.js \
  $(rg -l \"(cookie|session).*sameSite|secure|express-session|cookie-session\" src || true) \
; do
  if [ -f \"$f\" ]; then SESSION_FILE=\"$f\"; break; fi
done

if [ -n \"$SESSION_FILE\" ]; then
  echo \" - 세션 파일: $SESSION_FILE\"
  # sameSite 를 lax로, secure 를 true 로 강제 (존재하면 치환, 없으면 추가)
  # sameSite 치환/삽입
  if rg -n \"sameSite\\s*:\" \"$SESSION_FILE\" >/dev/null 2>&1; then
    perl -0777 -i -pe \"s/sameSite\\s*:\\s*[\\x27\\x22]?(none|lax|strict)[\\x27\\x22]?/sameSite: \\x27lax\\x27/ig\" \"$SESSION_FILE\"
  else
    perl -0777 -i -pe \"s/cookie\\s*:\\s*\\{\\s*/cookie: {\\n    sameSite: \\x27lax\\x27,\\n/s\" \"$SESSION_FILE\" || true
  fi
  # secure 치환/삽입
  if rg -n \"secure\\s*:\" \"$SESSION_FILE\" >/dev/null 2>&1; then
    perl -0777 -i -pe \"s/secure\\s*:\\s*(true|false|process\\.env\\.[^)\\s]+)/secure: true/ig\" \"$SESSION_FILE\"
  else
    perl -0777 -i -pe \"s/cookie\\s*:\\s*\\{\\s*/cookie: {\\n    secure: true,\\n/s\" \"$SESSION_FILE\" || true
  fi
  # path 보강
  if ! rg -n \"path\\s*:\" \"$SESSION_FILE\" >/dev/null 2>&1; then
    perl -0777 -i -pe \"s/cookie\\s*:\\s*\\{\\s*/cookie: {\\n    path: \\\\x27/\\\\x27,\\n/s\" \"$SESSION_FILE\" || true
  fi
else
  echo \" - 경고: 세션/쿠키 설정 파일을 못 찾았습니다. (JWT만 사용하는 구조일 수 있음)\"
fi

echo \"== 3) CORS/ORIGIN 허용 도메인 보강 =============\"
# allowedOrigins 배열에 운영 도메인 주입 (있으면 중복 방지)
CORS_FILE=\"\"
for f in src/middleware/cors.ts src/middleware/cors.js $(rg -l \"allowedOrigins|cors\\(\" src || true); do
  if [ -f \"$f\" ]; then CORS_FILE=\"$f\"; break; fi
done

if [ -n \"$CORS_FILE\" ]; then
  echo \" - CORS 파일: $CORS_FILE\"
  if ! rg -n \"$PROD_DOMAIN\" \"$CORS_FILE\" >/dev/null 2>&1; then
    # allowedOrigins 푸시 구문 주입 (배열이 존재한다고 가정)
    perl -0777 -i -pe \"s/(const\\s+allowedOrigins\\s*=\\s*[^;]+;)/\\1\\nif (!allowedOrigins.includes(\\x27$PROD_DOMAIN\\x27)) { allowedOrigins.push(\\x27$PROD_DOMAIN\\x27); }/s\" \"$CORS_FILE\" || true
  else
    echo \" - CORS: 이미 도메인 포함\"
  fi
else
  echo \" - 참고: 별도 CORS 미들웨어 파일을 찾지 못했습니다. (동일 오리진이면 문제 없음)\"
fi

echo \"== 4) 운영 시크릿(ENV) 세팅 =====================\"
replit secrets set NODE_ENV=production
replit secrets set SESSION_SECRET \"$(openssl rand -hex 32)\"
replit secrets set JWT_SECRET \"$(openssl rand -hex 32)\"
replit secrets set CORS_ORIGIN \"$PROD_DOMAIN\"
replit secrets set AUTH_ORIGIN \"$PROD_DOMAIN\"
# DATABASE_URL 은 사전에 올바르게 설정되어 있어야 합니다.

echo \"== 5) 운영 DB에 관리자 권한 보정 ===============\"
node - <<NODE
const { Client } = require(\"pg\");
const url = process.env.DATABASE_URL;
if (!url) { console.error(\"DATABASE_URL 이 없습니다.\"); process.exit(0); }
(async () => {
  const client = new Client({ connectionString: url });
  await client.connect();
  const email = \"$ADMIN_EMAIL\";
  try {
    // 1) 업데이트 시도 (가장 흔한 스키마: users(email, role, status, name, password_hash?) )
    let res = await client.query(\"UPDATE users SET role=\\'ADMIN\\', status=\\'ACTIVE\\' WHERE email=$1\", [email]);
    if (res.rowCount === 0) {
      // 2) 존재 안 하면 최소 컬럼으로 삽입 시도 (컬럼 불일치 시 catch)
      try {
        await client.query(\"INSERT INTO users(email, name, status, role) VALUES ($1, \\'Admin\\', \\'ACTIVE\\', \\'ADMIN\\')\", [email]);
        console.log(\" - 사용자 생성 및 ADMIN 부여: \", email);
      } catch (e) {
        console.warn(\" - 단순 INSERT 실패(스키마 상이). users 테이블 컬럼 구성이 다른 것으로 보입니다. 수동 보정 필요.\");
      }
    } else {
      console.log(\" - 기존 사용자 ADMIN 승격 완료\");
    }
  } catch (e) {
    console.warn(\" - users 테이블 업데이트 실패. 스키마가 다르거나 테이블명이 다릅니다. 수동 보정 필요.\");
  } finally {
    await client.end();
  }
})();
NODE

echo \"== 6) 빌드/마이그레이션/재배포 ===================\"
npm run db:migrate || echo \"(경고) db:migrate 실패 또는 스크립트 없음 — 스킵\"
npm run build || echo \"(경고) build 스크립트 없음 — 스킵\"
replit deploy --force

echo \"== 7) 헬스체크 & 로그인 플로우 점검 =============\"
set +e
echo \" - /api/auth/me (비로그인) 상태 확인\"
curl -sS -i \"$PROD_DOMAIN/api/auth/me\" | sed -n \"1,15p\"

echo \" - 로그인 시도 (Set-Cookie 확인)\"
LOGIN_RESP=$(curl -sS -i -X POST \"$PROD_DOMAIN/api/auth/login\" \
  -H \"Content-Type: application/json\" \
  --data \"{\\\"email\\\":\\\"$ADMIN_EMAIL\\\",\\\"password\\\":\\\"$ADMIN_PASSWORD\\\"}\" )
echo \"$LOGIN_RESP\" | sed -n \"1,25p\"
if echo \"$LOGIN_RESP\" | rg -i \"set-cookie\" >/dev/null 2>&1; then
  echo \" ✅ 서버에서 쿠키가 발급되었습니다 (Set-Cookie 감지).\"
else
  echo \" ⚠️ Set-Cookie 미감지 — 쿠키/프록시/도메인 설정 다시 확인 필요.\"
fi

echo \" - /api/auth/me (로그인 후) 재요청 (쿠키 자동 저장은 curl에서 안되므로 브라우저에서도 직접 확인 권장)\"
curl -sS -i \"$PROD_DOMAIN/api/auth/me\" | sed -n \"1,20p\"

echo \"== 완료: 운영 로그인 이슈 일괄 처리 시도 끝. 콘솔 출력 확인하세요. =="
'
