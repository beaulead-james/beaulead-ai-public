test -f client/public/favicon.ico || printf "dummy" > client/public/favicon.ico

echo "$RESP";
URL=$(node -e "console.log(JSON.parse(process.argv[1]).url)" "$RESP");
PUB=$(node -e "console.log(JSON.parse(process.argv[1]).publicUrl)" "$RESP");
echo "URL: $URL";
echo "PUBLIC: $PUB";
echo "--- HEAD check (relative) ---";
curl -sS -I "http://localhost:$PORT$URL" | head -n 1 || true;
echo "--- HEAD check (absolute) ---";
curl -sS -I "$PUB" | head -n 1 || true;
'
